CREATE TABLE `nvrs` (
	`id` INT(11) NOT NULL AUTO_INCREMENT,
	`dispositivo` VARCHAR(50) NOT NULL COLLATE 'utf8mb4_general_ci',
	`ip` VARCHAR(50) NOT NULL COLLATE 'utf8mb4_general_ci',
	`mac` VARCHAR(50) NOT NULL COLLATE 'utf8mb4_general_ci',
	`status` CHAR(1) NOT NULL DEFAULT '' COLLATE 'utf8mb4_general_ci',
	PRIMARY KEY (`id`) USING BTREE
)
COLLATE='utf8mb4_general_ci'
ENGINE=InnoDB
AUTO_INCREMENT=10
;



CREATE TABLE `cameras` (
	`id` INT(11) NOT NULL AUTO_INCREMENT,
	`id_nvr` INT(11) NOT NULL,
	`ip` VARCHAR(50) NOT NULL DEFAULT '' COLLATE 'utf8mb4_general_ci',
	`mac` VARCHAR(50) NOT NULL DEFAULT '' COLLATE 'utf8mb4_general_ci',
	`obs` TEXT NULL DEFAULT NULL COLLATE 'utf8mb4_general_ci',
	PRIMARY KEY (`id`) USING BTREE
)
COLLATE='utf8mb4_general_ci'
ENGINE=InnoDB
;


CREATE TABLE `camera_ping_failures` (
	`camera_key` VARCHAR(191) NOT NULL,
	`failed_attempts` TINYINT UNSIGNED NOT NULL DEFAULT 0,
	`last_checked` DATETIME NOT NULL,
	PRIMARY KEY (`camera_key`) USING BTREE
)
COLLATE='utf8mb4_general_ci'
ENGINE=InnoDB
;
